// =====================================================
// UDT_Fach  V1.1 — nur Fachdaten (keine Lager-Aggregate)
// In TIA Portal: PLC-Datentyp UDT_Fach entsprechend anpassen
// =====================================================
//
// Fachnummer          : USInt
// Materialart         : USInt
// Artikelnummer       : UInt
// ProductTyp          : UInt
// Kennzeichnung       : String[40]
// Datum eingelagert   : String[30]
// Datum_Ziffer        : DTL
// RFID_CODE           : UDInt
// Belegt              : Bool
// Gesperrt            : Bool
// Position_X          : Real
TYPE
    "UDT_Fach" : STRUCT
        Fachnummer        : USInt;       // 1..54
        Materialart       : USInt;
        Artikelnummer     : UInt;
        ProductTyp        : UInt;
        Kennzeichnung     : String[40];
        "Datum eingelagert" : String[30];
        Datum_Ziffer      : DTL;
        RFID_CODE         : UDInt;
        Belegt            : Bool;
        Gesperrt          : Bool;
        Position_X        : Real;        // bleibt beim Auslagern erhalten
        Position_Z        : Real;        // bleibt beim Auslagern erhalten
        RegalSeite        : USInt;       // 1=rechts, sonst links
        Inf_Text          : String[100]; // Fach-Notiz (nicht HMI Info_Text)
    END_STRUCT;
END_TYPE

//   Anzahl_Frei   : UInt
